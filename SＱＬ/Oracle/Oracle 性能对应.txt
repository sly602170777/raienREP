何时考虑追加索引？
追加索引的目标是减少全表扫描、加速数据定位。以下场景适合追加索引：
- 查询频繁使用某列作为过滤条件（WHERE）
- JOIN 操作中某列常被用作连接键
- ORDER BY 或 GROUP BY 中某列排序开销大
- 表数据量大，查询响应慢
- 执行计划显示全表扫描或高成本

**测试追加索引对性能的影响，建议采用“前后对比 + 执行计划分析 + 实时统计”的方法，结合查询响应时间、I/O行为和优化器路径选择进行验证。**

### 🔧 引用库

### 🧰 主模块代码

```csharp
public class OracleQueryOptimizer
{
    private readonly string _connectionString;

    public OracleQueryOptimizer(string connectionString)
    {
        _connectionString = connectionString;
    }

    public string GenerateOptimizedQuery(string tableName, string columnName, string filterValue)
    {
        string dataType = GetColumnDataType(tableName, columnName);
        if (dataType == null)
        {
            Console.WriteLine($"❌ 无法获取字段类型: {columnName}");
            return null;
        }

        string formattedValue = FormatValueByType(dataType, filterValue);
        string query = $"SELECT code FROM {tableName} WHERE {columnName} = {formattedValue}";
        Console.WriteLine($"✅ 最优查询语句: {query}");
        return query;
    }

    private string GetColumnDataType(string tableName, string columnName)
    {
        using var conn = new OracleConnection(_connectionString);
        conn.Open();

        string sql = @"
            SELECT data_type 
            FROM all_tab_columns 
            WHERE table_name = :tableName AND column_name = :columnName";

        using var cmd = new OracleCommand(sql, conn);
        cmd.Parameters.Add(new OracleParameter("tableName", tableName.ToUpper()));
        cmd.Parameters.Add(new OracleParameter("columnName", columnName.ToUpper()));

        using var reader = cmd.ExecuteReader();
        return reader.Read() ? reader.GetString(0) : null;
    }

    private string FormatValueByType(string dataType, string value)
    {
        return dataType switch
        {
            "VARCHAR2" => $"'{value}'",
            "CHAR"     => $"'{value}'",
            "NUMBER"   => value,
            "DATE"     => $"TO_DATE('{value}', 'YYYYMMDD')",
            _          => $"'{value}'" // 默认按字符串处理
        };
    }
}
```
